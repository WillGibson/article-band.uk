{{- /* Simple audio player with an optional download link. Usage: {{< audio src="audio/my-track-128kbps.mp3" download="audio/my-track-320kbps.mp3" title="My track" >}} {{< audio src="audio/my-track-128kbps.mp3" download="none" title="My track" >}} `src` is the file the player streams. `download` is the file the download link offers (defaults to `src`); use "none" to hide the link. Paths are looked up in the current page's bundle first, then in the site root (content/, e.g. content/audio/), and otherwise used as a URL. */ -}} {{- $page := .Page -}} {{- $src := .Get "src" -}} {{- $download := .Get "download" | default $src -}} {{- $title := .Get "title" | default (path.BaseName $src) -}} {{- $urls := dict -}} {{- range $key, $path := dict "src" $src "download" $download -}} {{- $url := $path -}} {{- with or ($page.Resources.Get $path) (site.Home.Resources.Get $path) -}} {{- $url = .RelPermalink -}} {{- end -}} {{- $urls = merge $urls (dict $key $url) -}} {{- end -}}
